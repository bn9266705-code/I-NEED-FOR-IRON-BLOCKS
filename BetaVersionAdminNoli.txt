AN.LoadAnimFromUrl = function(animName, url)
    local content = AN.httpGet(url)
    if not content or content == "" then
        AN.warn("HttpGet failed for anim: " .. animName)
        return false
    end
    local path = "AdminNoli_" .. animName .. ".rbxm"
    local wok = pcall(writefile, path, content)
    if not wok then
        AN.warn("writefile failed for anim: " .. animName)
        return false
    end
    local aok, assetId = pcall(getcustomasset, path)
    if not aok or not assetId then
        AN.warn("getcustomasset failed for anim: " .. animName)
        return false
    end
    local kfSeq
    local okObj, resObj = pcall(game.GetObjects, game, assetId)
    if okObj and resObj and #resObj > 0 then
        for _, obj in ipairs(resObj) do
            if obj:IsA("KeyframeSequence") then kfSeq = obj break end
            for _, d in ipairs(obj:GetDescendants()) do
                if d:IsA("KeyframeSequence") then kfSeq = d break end
            end
            if kfSeq then break end
        end
    else
        local insert = game:GetService("InsertService")
        local okIns, inserted = pcall(function() return insert:LoadLocalAsset(assetId) end)
        if okIns and inserted then
            if inserted:IsA("KeyframeSequence") then kfSeq = inserted end
            if not kfSeq then
                for _, d in ipairs(inserted:GetDescendants()) do
                    if d:IsA("KeyframeSequence") then kfSeq = d break end
                end
            end
        end
    end
    if not kfSeq then
        AN.warn("KeyframeSequence not found for: " .. animName)
        return false
    end
    local kfs = {}
    local maxT = 0
    for _, kf in ipairs(kfSeq:GetKeyframes()) do
        local poseData = {}
        local function extract(container)
            for _, pose in ipairs(container:GetChildren()) do
                if pose:IsA("Pose") then
                    poseData[pose.Name] = pose.CFrame
                    extract(pose)
                end
            end
        end
        extract(kf)
        table.insert(kfs, {Time = kf.Time, Poses = poseData})
        if kf.Time > maxT then maxT = kf.Time end
    end
    table.sort(kfs, function(a, b) return a.Time < b.Time end)
    AN.KeyframeData[animName] = kfs
    AN.AnimDurations[animName] = maxT
    return true
end

AN.setupSoundReplacer = function()
    local function tagOnce(inst)
        if inst:FindFirstChild("AdminNoliReplaced") then return false end
        local tag = Instance.new("BoolValue")
        tag.Name = "AdminNoliReplaced"
        tag.Parent = inst
        return true
    end
    local function checkSound(inst)
        if not inst:IsA("Sound") then return end
        local function replace()
            if inst:FindFirstChild("IsAdminSound") then return end
            if inst:FindFirstChild("AdminNoliReplaced") then return end
            local id = inst.SoundId
            local match = id:match("rbxassetid://(%d+)") or id:match("id=(%d+)")
            if match then id = "rbxassetid://" .. match end
            if AN.KILL_VL_IDS[id] then
                local recent = AN.lastVLPlayed["Kill"] and (os.clock() - AN.lastVLPlayed["Kill"]) < 0.5
                if recent then
                    tagOnce(inst)
                    inst.Volume = 0
                    return
                end
                local customKill = AN.pickCustomKillSound()
                if customKill then
                    tagOnce(inst)
                    inst.SoundId = customKill
                    inst.Volume = 1.5
                    local tag = Instance.new("BoolValue")
                    tag.Name = "IsAdminSound"
                    tag.Parent = inst
                    AN.lastVLPlayed["Kill"] = os.clock()
                    AN.stopSound("Idle")
                    AN.lastVLTime = os.clock()
                end
                return
            end
            if AN.NOLI_ORIGINAL_VLS[id] or AN.MUTE_SET[id] then
                tagOnce(inst)
                inst.Volume = 0
            end
            local repl = AN.SOUND_REPLACER[id]
            if repl and AN.SoundData[repl.customSound] then
                tagOnce(inst)
                inst.SoundId = AN.SoundData[repl.customSound]
                inst.Volume = repl.volume or 1.5
                local tag = Instance.new("BoolValue")
                tag.Name = "IsAdminSound"
                tag.Parent = inst
                if repl.looped ~= nil then inst.Looped = repl.looped end
                if AN.STOP_IDLE_SOUNDS[repl.customSound] then AN.stopSound("Idle") end
            end
            local vlCategory = AN.VL_MAP[id]
            if vlCategory then
                tagOnce(inst)
                inst.Volume = 0
                AN.playVoiceline(vlCategory)
            end
        end
        replace()
        inst:GetPropertyChangedSignal("SoundId"):Connect(replace)
        inst.Played:Connect(replace)
    end
    for _, obj in ipairs(game:GetDescendants()) do checkSound(obj) end
    game.DescendantAdded:Connect(checkSound)
end

AN.ProcessarKeyframesEmTempoReal = function(dt)
    if not AN.currentAnimName or not AN.KeyframeData[AN.currentAnimName] or #AN.KeyframeData[AN.currentAnimName] == 0 then return end
    local kfList = AN.KeyframeData[AN.currentAnimName]
    local maxTime = AN.AnimDurations[AN.currentAnimName] or 1
    AN.animTime = AN.animTime + (dt * AN.CONFIG.VelocidadeGlobal)
    if AN.animTime >= maxTime then
        if AN.isLoopingAnim then
            AN.animTime = AN.animTime % maxTime
        else
            AN.animTime = maxTime
            if AN.activeOverride then
                if AN.activeOverride == "ObservantLoop" and AN.KeyframeData["ObservantIdle"] then
                    if AN.hasActiveObservantServerTrack() then
                        AN.iniciarOverride("ObservantIdle", true, false)
                        return
                    end
                end
                if AN.overrideClearOnEnd then AN.limparOverride() return end
            else
                AN.currentAnimName = ""
                return
            end
        end
    end
    local kfPrev = kfList[1]
    local kfNext = kfList[#kfList]
    for i = 1, #kfList - 1 do
        if AN.animTime >= kfList[i].Time and AN.animTime <= kfList[i+1].Time then
            kfPrev = kfList[i]
            kfNext = kfList[i+1]
            break
        end
    end
    local span = kfNext.Time - kfPrev.Time
    local alpha = span > 0 and math.clamp((AN.animTime - kfPrev.Time) / span, 0, 1) or 0
    for jointName, joint in pairs(AN.jointMap) do
        if joint and joint.Parent then
            local poseA = kfPrev.Poses[jointName]
            local poseB = kfNext.Poses[jointName]
            if poseA and poseB then
                joint.Transform = poseA:Lerp(poseB, alpha)
            elseif poseA then
                joint.Transform = poseA
            elseif poseB then
                joint.Transform = poseB
            else
                joint.Transform = CFrame.identity
            end
        end
    end
end

        return
    end
    local entry = AN.ANIM_MAP[id]

    if entry.name == "IGNORE" then return end
    
    if entry.name == "Nova" then
        AN.playVoiceline("NovaThrow")
    elseif entry.name == "ExecKiller" then
        AN.playVoiceline("Kill")

        AN.stopSound("VoidRushing")
        AN.playVoiceline("VoidRushCrash")
    elseif entry.name == "VoidRushHit" or entry.name == "VoidRushCutscene" or entry.name == "VoidRushEnd" or entry.name == "VoidRushCancel" then
        AN.stopSound("VoidRushing")
    end
    
    if entry.name == "VoidRushHit" or entry.name == "VoidRushCrash" or entry.name == "VoidRushCutscene" or entry.name == "VoidRushEnd" or entry.name == "VoidRushCancel" then
        if AN.vrDashSound then AN.vrDashSound:Stop() AN.vrDashSound:Destroy() AN.vrDashSound = nil end
    end

AN.handleVoidRushState = function(rig, newState)
    if newState == AN.currentVoidRushState then return end
    local oldState = AN.currentVoidRushState
    AN.currentVoidRushState = newState
    if newState ~= "Dashing" and AN.vrDashSound then
        AN.vrDashSound:Stop() AN.vrDashSound:Destroy() AN.vrDashSound = nil
    end
    if newState == "Charging" then
        if AN.KeyframeData["VoidRushLoop"] then AN.iniciarOverride("VoidRushLoop", true, false) end
    elseif newState == "Dashing" then
        if AN.KeyframeData["VoidRushDash"] then
            AN.iniciarOverride("VoidRushDash", true, false)
            if not AN.vrDashSound and AN.SoundData["VoidRushVrooom"] then
                AN.vrDashSound = Instance.new("Sound")
                AN.vrDashSound.SoundId = AN.SoundData["VoidRushVrooom"]
                AN.vrDashSound.Volume = 1.5
                AN.vrDashSound.Looped = true
                local tag = Instance.new("BoolValue")
                tag.Name = "IsAdminSound"
                tag.Parent = AN.vrDashSound
                AN.vrDashSound.Parent = rig:FindFirstChild("HumanoidRootPart") or rig
                AN.vrDashSound:Play()
            end
        elseif AN.KeyframeData["Run"] then
            AN.iniciarOverride("Run", true, false)
        end
    elseif newState == "ChargingCancelled" then
        AN.stopSound("VoidRushing")
        if AN.KeyframeData["VoidRushCancel"] then AN.iniciarOverride("VoidRushCancel", false, true) end
    elseif (newState == nil or newState == "") and (oldState == "Charging" or oldState == "ChargingCancelled") then
        AN.stopSound("VoidRushing")
        if oldState == "Charging" and AN.KeyframeData["VoidRushCancel"] then
            AN.iniciarOverride("VoidRushCancel", false, true)
        else
            AN.limparOverride()
        end
    elseif newState == nil or newState == "" then
        AN.limparOverride()
    end
end

getgenv().AN_VisualRender = RunService.PreRender:Connect(function(dt)
    local corpo = Player.Character
    if not corpo or not corpo.Parent then
        if AN.VisualModel and AN.VisualModel.Parent then AN.VisualModel.Parent = nil end
        return
    end
    if corpo ~= AN.corpoInicial then
        AN.setupCharacter(corpo)
        task.spawn(function() task.wait(0.5) AN.hookAbilityAnimations(corpo) end)
        return
    end
    local humanoid = corpo:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then
        if AN.VisualModel and AN.VisualModel.Parent then AN.VisualModel.Parent = nil end
        return
    end
    
    local isNoli = corpo.Name == "Noli"
    local skin = corpo:GetAttribute("SkinName")
    local semSkin = (skin == nil or skin == "")
    
    if not isNoli or not semSkin then
        if AN.VisualModel and AN.VisualModel.Parent then AN.VisualModel.Parent = nil end
        return
    end

    if AN.VisualModel and not AN.VisualModel.Parent then
        AN.VisualModel.Parent = workspace
        AN.lastState = ""
    end
    
    AN.esconderPartesOriginais(corpo)

    local atualRoot = corpo:FindFirstChild("HumanoidRootPart")
    if not atualRoot or not AN.rigRoot or not AN.modeloSelecionado then return end
    
    humanoid.CameraOffset = AN.CONFIG.OffsetCamera
    local rootCFrame = atualRoot.CFrame
    local _, ry, _ = rootCFrame:ToOrientation()
    AN.rigRoot.CFrame = CFrame.new(rootCFrame.Position) * CFrame.Angles(0, ry, 0) * AN.CONFIG.OffsetModelo
    
    if not AN.activeOverride then
        local velReal = atualRoot.AssemblyLinearVelocity
        local velPlana = math.sqrt(velReal.X^2 + velReal.Z^2)
        local ws = humanoid.WalkSpeed
        local desired
        
        if velPlana > 1.8 then
            if velPlana >= AN.CONFIG.LimiteVelocidadeCorrida or ws >= AN.CONFIG.LimiteVelocidadeCorrida then
                desired = "Run"
            else
                desired = "Walk"
            end
        else
            desired = "Idle"
        end
        
        if AN.lastState ~= desired and AN.KeyframeData[desired] then
            AN.lastState = desired
            AN.PlayAnimationRealtime(desired, false)
        end
    end
    AN.ProcessarKeyframesEmTempoReal(dt)
end)
